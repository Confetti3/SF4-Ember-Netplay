cmake_minimum_required(VERSION 3.16)
project(EmberUiPolishTests LANGUAGES CXX)
enable_testing()
add_executable(EmberUiPolishTests ui_polish_test.cxx)
target_compile_features(EmberUiPolishTests PRIVATE cxx_std_14)
if(MSVC)
    target_compile_options(EmberUiPolishTests PRIVATE /W4)
else()
    target_compile_options(EmberUiPolishTests PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
endif()
add_test(NAME EmberUiPolish COMMAND EmberUiPolishTests)
set_tests_properties(EmberUiPolish PROPERTIES TIMEOUT 10)
