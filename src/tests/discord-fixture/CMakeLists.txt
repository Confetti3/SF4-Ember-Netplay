cmake_minimum_required(VERSION 3.20)
project(DiscordBridgeFixture LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Bridge fixture must exercise an x64 companion")
endif()
add_executable(DiscordBridgeFixture ../discord_bridge_fixture.cxx)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/sf4e_product.cmake")
sf4e_add_product_identity(DiscordBridgeFixture)
target_compile_features(DiscordBridgeFixture PRIVATE cxx_std_17)
target_compile_definitions(DiscordBridgeFixture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
set_property(TARGET DiscordBridgeFixture PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_link_libraries(DiscordBridgeFixture PRIVATE advapi32)
