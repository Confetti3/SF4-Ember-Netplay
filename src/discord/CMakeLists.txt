cmake_minimum_required(VERSION 3.20)
if(NOT SF4E_PRODUCT_VERSION)
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../CMakeLists.txt" version_line REGEX "^[ ]+VERSION [0-9]+[.][0-9]+[.][0-9]+$")
    string(STRIP "${version_line}" version_line)
    string(REPLACE "VERSION " "" SF4E_PRODUCT_VERSION "${version_line}")
endif()
project(EmberDiscord VERSION ${SF4E_PRODUCT_VERSION} LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Ember Discord requires a separate Windows x64 build")
endif()
if(NOT SF4E_DISCORD_APPLICATION_ID MATCHES "^[1-9][0-9]+$")
    message(FATAL_ERROR "Set the public Ember Discord Application ID")
endif()
foreach(file include/discordpp.h lib/release/discord_partner_sdk.lib bin/release/discord_partner_sdk.dll)
    if(NOT EXISTS "${SF4E_DISCORD_SDK_ROOT}/${file}")
        message(FATAL_ERROR "Missing pinned Discord SDK file: ${file}")
    endif()
endforeach()
add_executable(ember-discord WIN32 main.cxx)
set(SF4E_BRAND_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(SF4E_VERSION_MAJOR "${PROJECT_VERSION_MAJOR}")
set(SF4E_VERSION_MINOR "${PROJECT_VERSION_MINOR}")
set(SF4E_VERSION_PATCH "${PROJECT_VERSION_PATCH}")
set(SF4E_FILE_DESCRIPTION "SF4 Ember Netplay Discord companion")
set(SF4E_ORIGINAL_FILENAME "ember-discord.exe")
set(SF4E_INTERNAL_NAME "ember-discord.exe")
set(SF4E_FILE_TYPE "VFT_APP")
configure_file("${SF4E_BRAND_SOURCE_DIR}/cmake/version.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/ember-discord.rc" @ONLY)
target_sources(ember-discord PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ember-discord.rc")
target_compile_features(ember-discord PRIVATE cxx_std_20)
set_property(TARGET ember-discord PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(ember-discord PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
    SF4E_DISCORD_APPLICATION_ID=${SF4E_DISCORD_APPLICATION_ID}ULL)
target_include_directories(ember-discord PRIVATE "${SF4E_DISCORD_SDK_ROOT}/include" "${SF4E_JSON_INCLUDE}")
target_link_libraries(ember-discord PRIVATE "${SF4E_DISCORD_SDK_ROOT}/lib/release/discord_partner_sdk.lib" advapi32 shell32)
install(TARGETS ember-discord RUNTIME DESTINATION .)
install(FILES "${SF4E_DISCORD_SDK_ROOT}/bin/release/discord_partner_sdk.dll" DESTINATION .)
